18const std::vector<double> BezierInterpolate(
19 const std::vector<double> & y, const std::vector<unsigned int> & index
20)
21{
22 struct point {
23 double x;
24 double y;
25 };
26
27 std::vector<point> p;
28 std::vector<double> z(y.size());
29
30 p.push_back({(double)index[0], y[index[0]]});
31
32 for(unsigned int i = 1; i < index.size() - 2; i++) {
33 p.push_back({(double)index[i], y[index[i]]});
34 p.push_back({(double)(index[i] + index[i + 1]) / 2, (y[index[i]] + y[index[i + 1]]) / 2});
35 }
36
37 p.push_back({(double)index[index.size() - 2], y[index[index.size() - 2]]});
38 p.push_back({(double)index[index.size() - 1], y[index[index.size() - 1]]});
39
40 for(unsigned int i = 0, j = 0; j < p.size() - 2; j += 2) {
41 double a = p[j].x - 2 * p[j + 1].x + p[j + 2].x;
42 double b = -2 * p[j].x + 2 * p[j + 1].x;
43
44 while((double)i <= p[j + 2].x) {
45 double c = p[j].x - (double)i;
46 double t = (-b + std::sqrt(b * b - 4 * a * c)) / (2 * a);
47
48 z[i] = (1 - t) * (1 - t) * p[j].y + 2 * (1 - t) * t * p[j + 1].y + t * t * p[j + 2].y;
49 i++;
50 }
51 }
52
53 return z;
54}

67 throw std::invalid_argument("BaselineCornerCutting(): the length of y is zero.");

75 std::vector<double> er_list;
76 double a_old = 0;
77 unsigned int marked_size_old = y.size();
78
79 for(unsigned int j = 0; j < y.size() - 1; j++) {
80 a_old += (y[j + 1] + y[j]);
81 }
82
83 a_old /= 2;
84
85 for(unsigned int i = 0; i < loop; i++) {
86 std::vector<unsigned int> p;
87
88 for(unsigned int j = 0; j < t.size(); j++) {
89 if(t[j] == i) {
90 p.push_back(j);
91 }
92 }
93
94 std::vector<unsigned int> marked;
95 unsigned int marked_size_new;
96 unsigned int elmininated_count = 0;
97
98 marked.push_back(p.front());
99 t[p.front()]++;
100
101 for(unsigned int j = 1; j < p.size() - 1; j++) {
102 double c = (y[p[j + 1]] - y[p[j - 1]]) / (p[j + 1] - p[j - 1]) * (p[j] - p[j - 1]) + y[p[j - 1]];
103
104 if(y[p[j]] < c) {
105 marked.push_back(p[j]);
106 t[p[j]]++;
107 }
108 }
109
110 marked.push_back(p.back());
111 t[p.back()]++;
112
113 marked_size_new = marked.size();
114 elmininated_count = marked_size_old - marked_size_new;
115
116 if(marked_size_new == 0 || elmininated_count == 0) {
117 break;
118 }
119
120 // Note:
121 // A straightforward reading of the Terminal Condition in Liu's paper suggests
122 // that the area should be calculated using the remaining points. However,
123 // Algorithm 2 can also be interpreted as calculating the area using the
124 // removed points. Furthermore, the definition of ER is also different.
125 // In this implementation, we give priority to the description in the
126 // Terminal Condition and calculate the area using the remaining points.
127 double a_new = 0;
128
129 for(unsigned int j = 0; j < marked.size() - 1; j++) {
130 a_new += ((double)marked[j + 1] - (double)marked[j]) * (y[marked[j + 1]] + y[marked[j]]);
131 }
132
133 a_new /= 2;
134
135 // Calculate ER value.
136 double er = (a_old - a_new) / elmininated_count;
137 er_list.push_back(er);
138 a_old = a_new;
139 marked_size_old = marked_size_new;
140 }
141
142 // Search for maximum ER value
143 unsigned int max_er_index = std::max_element(er_list.begin(), er_list.end()) - er_list.begin();
144
145 // Make index list
146 std::vector<unsigned int> index;
147
148 // Note:
149 // We search for the maximum ER value, but note that the points used to
150 // calculate the ER value (i.e., the points used to construct the Bézier
151 // curve) are those before T is incremented.
152 // Therefore, the point corresponding to the maximum ER value can only be
153 // identified by using the ER value from the previous iteration.
154 for(unsigned int i = 0; i < t.size(); i++) {
155 if(t[i] >= max_er_index - 1) {
156 index.push_back(i);
157 }
158 }
159
160 // Corner-Cutting with Bezier technique
161 std::vector<double> z = BezierInterpolate(y, index);
162
163 BaselineResult result;
164 result.baseline.resize(z.size());
165 result.corrected.resize(z.size());
166
167 for(unsigned int i = 0; i < z.size(); i++) {
168 result.baseline[i] = z[i];
169 result.corrected[i] = y[i] - z[i];
170 }
171
172 return result;
173}
