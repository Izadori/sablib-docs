58 alpha.setOnes(m);

61 I.setIdentity();

63 lambdaDTD = lambda * (D.transpose() * D);
64
65 for(unsigned int i = 0; i < loop; i++) {
66 A = alpha.asDiagonal();

72

75 sd = std::sqrt((d.array() < 0).select((d.array() - mean).matrix(), 0).squaredNorm() / (ct - 1));
76
77 wt = (-k * (d.array() - sd) / sd).matrix().unaryExpr(std::ref(logistic));
78
79 if ((w - wt).norm() / w.norm() < eps) {
80 break;
81 }

84 }
